find_package(Eigen3 CONFIG REQUIRED)

function(ksj_target_enable_benchmark_defaults target_name)
  ksj_target_enable_defaults(${target_name})
  target_link_libraries(${target_name} PRIVATE Eigen3::Eigen)
  if(TARGET KSpaceJet::numerics_runtime)
    target_link_libraries(${target_name} PRIVATE KSpaceJet::numerics_runtime)
  endif()
  ksj_install_target(${target_name})
  ksj_register_test_suite_executable(BENCHMARKS ${target_name})
endfunction()
